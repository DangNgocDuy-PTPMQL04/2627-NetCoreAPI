{
    private readonly RoleManager<IdentityRole> _roleManager;
    private readonly UserManager<DemoMVC.Models.ApplicationUser> _userManager;

    public RoleController(
        RoleManager<IdentityRole> roleManager,
        UserManager<DemoMVC.Models.ApplicationUser> userManager)
    {
        _roleManager = roleManager;
        _userManager = userManager;
    }

    // Danh sÃ¡ch Role
    [HttpGet]
    public async Task<IActionResult> Index()
    {
        var roles = await _roleManager.Roles
            .OrderBy(r => r.Name)
            .ToListAsync();

        var result = new List<RoleViewModel>();

        foreach (var role in roles)
        {
            result.Add(new RoleViewModel
            {
                Id = role.Id,
                Name = role.Name ?? ""
            });
        }

        ViewBag.UserCounts = new Dictionary<string, int>();

        foreach (var role in roles)
        {
            var users = await _userManager.GetUsersInRoleAsync(
                role.Name ?? ""
            );

            ViewBag.UserCounts[role.Id] = users.Count;
        }

        return View(result);
    }

                .Where(s =>

using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

namespace DemoMVC.Controllers;

[Authorize(Roles = "Admin")]
public class UsersController : Controller
{
    private readonly UserManager<ApplicationUser> _userManager;
    private readonly RoleManager<IdentityRole> _roleManager;

    public UsersController(
        UserManager<ApplicationUser> userManager,
        RoleManager<IdentityRole> roleManager)
    {
        _userManager = userManager;
        _roleManager = roleManager;
    }

    // Danh sÃ¡ch tÃ i khoáº£n
    [HttpGet]
    public async Task<IActionResult> Index()
    {
        var users = await _userManager.Users
            .OrderBy(u => u.Email)
            .ToListAsync();

        var result = new List<UserRoleViewModel>();

        foreach (var user in users)
        {
            var roles = await _userManager.GetRolesAsync(user);

            result.Add(new UserRoleViewModel
            {
                UserId = user.Id,
                Email = user.Email ?? "",
                FullName = user.FullName ?? "",
                Roles = roles.Select(r => new RoleSelectionViewModel
                {
                    RoleName = r,
                    IsSelected = true
                }).ToList()
            });
        }

        return View(result);
    }

    // Trang gÃ¡n Role
    [HttpGet]
    public async Task<IActionResult> ManageRoles(string id)
    {
        if (string.IsNullOrWhiteSpace(id))
            return NotFound();

        var user = await _userManager.FindByIdAsync(id);

        if (user == null)
            return NotFound();

        var userRoles = await _userManager.GetRolesAsync(user);

        var allRoles = await _roleManager.Roles
            .OrderBy(r => r.Name)
            .ToListAsync();

        var model = new UserRoleViewModel
        {
            UserId = user.Id,
            Email = user.Email ?? "",
            FullName = user.FullName ?? "",
            Roles = allRoles.Select(role =>
                new RoleSelectionViewModel
                {
                    RoleName = role.Name ?? "",
                    IsSelected = userRoles.Contains(role.Name ?? "")
                }).ToList()
        };

        return View(model);
    }

    // LÆ°u Role
    [HttpPost]
    [ValidateAntiForgeryToken]
    public async Task<IActionResult> ManageRoles(UserRoleViewModel model)
    {
        var user = await _userManager.FindByIdAsync(model.UserId);

        if (user == null)
            return NotFound();

        var selectedRoles = model.Roles
            .Where(r => r.IsSelected)
            .Select(r => r.RoleName)
            .Distinct(StringComparer.OrdinalIgnoreCase)
            .ToList();

        // Chá»‰ cho phÃ©p gÃ¡n nhá»¯ng Role Ä‘ang tá»“n táº¡i.
        foreach (var roleName in selectedRoles)
        {
            if (!await _roleManager.RoleExistsAsync(roleName))
            {
                TempData["Error"] = $"Role '{roleName}' khÃ´ng tá»“n táº¡i.";
                return RedirectToAction(nameof(ManageRoles),
                    new { id = model.UserId });
            }
        }

        var currentRoles = await _userManager.GetRolesAsync(user);
        bool wasAdmin = currentRoles.Contains("Admin");
        bool willBeAdmin = selectedRoles.Contains(
            "Admin", StringComparer.OrdinalIgnoreCase);

        // KhÃ´ng cho tá»± thu há»“i quyá»n Admin.
        if (user.Id == _userManager.GetUserId(User)
            && wasAdmin && !willBeAdmin)
        {
            TempData["Error"] =
                "Báº¡n khÃ´ng thá»ƒ tá»± thu há»“i quyá»n Admin cá»§a chÃ­nh mÃ¬nh.";

            return RedirectToAction(nameof(ManageRoles),
                new { id = model.UserId });
        }

        // KhÃ´ng cho xÃ³a quyá»n Admin cuá»‘i cÃ¹ng cá»§a há»‡ thá»‘ng.
        if (wasAdmin && !willBeAdmin)
        {
            var adminUsers = await _userManager.GetUsersInRoleAsync("Admin");

            if (adminUsers.Count <= 1)
            {
                TempData["Error"] =
                    "KhÃ´ng thá»ƒ thu há»“i quyá»n Admin cuá»‘i cÃ¹ng cá»§a há»‡ thá»‘ng.";

                return RedirectToAction(nameof(ManageRoles),
                    new { id = model.UserId });
            }
        }

        var rolesToRemove = currentRoles
            .Where(r => !selectedRoles.Contains(
                r, StringComparer.OrdinalIgnoreCase))
            .ToList();

        var rolesToAdd = selectedRoles
            .Where(r => !currentRoles.Contains(
                r, StringComparer.OrdinalIgnoreCase))
            .ToList();

        if (rolesToRemove.Count > 0)
        {
            var removeResult = await _userManager.RemoveFromRolesAsync(
                user, rolesToRemove);

            if (!removeResult.Succeeded)
            {
                TempData["Error"] = string.Join(
                    ", ",
                    removeResult.Errors.Select(e => e.Description));

                return RedirectToAction(nameof(ManageRoles),
                    new { id = model.UserId });
            }
        }

        if (rolesToAdd.Count > 0)
        {
            var addResult = await _userManager.AddToRolesAsync(
                user, rolesToAdd);

            if (!addResult.Succeeded)
            {
                TempData["Error"] = string.Join(
                    ", ",
                    addResult.Errors.Select(e => e.Description));

                return RedirectToAction(nameof(ManageRoles),
                    new { id = model.UserId });
            }
        }

        TempData["Success"] = "Cáº­p nháº­t vai trÃ² ngÆ°á»i dÃ¹ng thÃ nh cÃ´ng.";

        return RedirectToAction(nameof(Index));
    }
}
